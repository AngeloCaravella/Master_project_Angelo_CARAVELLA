\section{RL Algorithms}
This work benchmarks several state-of-the-art DRL algorithms. The following sections provide a detailed mathematical description of the selected off-policy, actor-critic algorithms that form the core of the experimental comparison.

\subsection{Soft Actor-Critic}
SAC is an off-policy actor-critic algorithm designed for \textbf{continuous action spaces} that optimizes a stochastic policy. Its distinguishing feature is the explicit incorporation of entropy maximization into the objective function, which promotes exploration and yields policies that are more robust. Unlike traditional RL algorithms that focus solely on reward maximization, SAC aims to maximize both the expected cumulative reward and the entropy of the policy distribution.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\linewidth]{SAC.png}
    \caption{SAC architecture.}
    \label{fig:SAC}
\end{figure}

The objective function is:
\[
J(\pi) = \sum_{t=0}^{T} \mathbb{E}_{(s_t, a_t) \sim \rho_\pi} \left[ r(s_t, a_t) + \alpha \mathcal{H}(\pi(\cdot|s_t)) \right]
\]
where $\mathcal{H}$ is the entropy of the policy $\pi$, and $\alpha$ is the temperature parameter, which controls the trade-off between reward exploitation and entropy-driven exploration.

\begin{algorithm}[H]
\caption{Soft Actor-Critic}
\label{alg:sac}
\begin{algorithmic}[1]
\State Initialize critic networks $Q_{\theta_1}, Q_{\theta_2}$, actor network $\pi_\phi$, and target networks.
\State Initialize replay buffer $D$ and temperature $\alpha$.
\For{each timestep $t=1, \dots, T$}
    \State Select action $a_t \sim \pi_\phi(\cdot|s_t)$, execute, and store transition in $D$.
    \If{update step}
        \State Sample a minibatch from $D$.
        \State Compute Q-function target $y_j$ using target networks and entropy term.
        \State Update critic networks by minimizing MSE loss w.r.t. $y_j$.
        \State Update actor network by maximizing the policy objective.
        \State (Optional) Update temperature $\alpha$ to target a specific entropy.
        \State Update target networks using polyak averaging.
    \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}

SAC employs a soft Q-function, trained to minimize the soft Bellman residual. The soft state value function $V$ is defined as:
\[
V_{\text{soft}}(s_t) = \mathbb{E}_{a_t \sim \pi} [Q_{\text{soft}}(s_t, a_t) - \alpha \log \pi(a_t|s_t)]
\]
To mitigate the positive bias inherent in Q-learning, SAC employs two independent Q-networks (Clipped Double-Q technique) and uses the minimum of the two target Q-values during the Bellman update. Additionally, SAC can automatically tune the temperature parameter $\alpha$ during training.

\subsection{Deep Deterministic Policy Gradient + PER}
DDPG is an off-policy algorithm that concurrently learns a deterministic policy $\mu(s | \theta^\mu)$ and a Q-function $Q(s, a | \theta^Q)$. It extends the Deterministic Policy Gradient (DPG) theorem to DNNs, enabling continuous control in high-dimensional state spaces.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.5\linewidth]{DDPG.png}
    \caption{DDPG architecture.}
    \label{fig:DDPG}
\end{figure}

\begin{algorithm}[H]
\caption{DDPG with Prioritized Experience Replay (PER)}
\label{alg:ddpg_per}
\begin{algorithmic}[1]
\State Initialize critic $Q$, actor $\mu$, and their target networks $Q', \mu'$.
\State Initialize Prioritized Replay Buffer $D$.
\For{each episode}
    \For{each timestep $t=1, \dots, T$}
        \State Select action $a_t = \mu(s_t | \theta^\mu) + \mathcal{N}_t$ and store transition in $D$.
        \State Sample minibatch from $D$ using priorities; compute IS weights.
        \State Compute targets: $y_j = r_j + \gamma Q'(s_{j+1}, \mu'(s_{j+1}))$.
        \State Compute TD-errors $\delta_j$ and update critic by minimizing weighted loss.
        \State Update transition priorities in $D$ with $|\delta_j|$.
        \State Update actor using the sampled policy gradient.
        \State Update target networks using polyak averaging.
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{itemize}
    \item \textbf{Critic Update:} The critic is updated by minimizing the mean-squared Bellman error. Target networks ($Q'$ and $\mu'$) are used to stabilize training. The target $y_t$ is given by:
    \[
    y_t = r_t + \gamma Q'(s_{t+1}, \mu'(s_{t+1}|\theta^{\mu'})|\theta^{Q'})
    \]
    
    \item \textbf{Actor Update:} The actor is updated using the deterministic policy gradient theorem:
    \[
    \nabla_{\theta^\mu} J \approx \mathbb{E}_{s_t \sim D} [\nabla_a Q(s, a | \theta^Q)|_{s=s_t, a=\mu(s_t)} \nabla_{\theta^\mu} \mu(s_t | \theta^\mu)]
    \]
    
    \item \textbf{PER:} PER samples transitions according to their temporal-difference (TD) error, prioritizing experiences where the model's predictions are most inaccurate. The probability of sampling transition $i$ is $P(i) = \frac{p_i^\beta}{\sum_k p_k^\beta}$. To correct for the bias introduced by non-uniform sampling, PER applies importance-sampling (IS) weights during the gradient update.
\end{itemize}

\subsection{Truncated Quantile Critics}
TQC represents a significant advancement over standard SAC by modeling the entire distribution of returns rather than just its expected value. This distributional perspective, combined with a novel truncation mechanism, provides superior protection against Q-value overestimation.

\begin{algorithm}[H]
\caption{Truncated Quantile Critics (TQC)}
\label{alg:tqc}
\begin{algorithmic}[1]
\State Initialize critic ensemble $\{Q_{\theta_i}\}_{i=1}^N$, actor $\pi_\phi$, and target networks.
\State Initialize replay buffer $D$. Set number of quantiles $N$ and number to drop $k$.
\For{each timestep $t=1, \dots, T$}
    \State Select action $a_t \sim \pi_\phi(\cdot|s_t)$, execute, and store transition in $D$.
    \If{update step}
        \State Sample a minibatch from $D$.
        \State Get next-state quantile estimates from target critics.
        \State Sort estimates and drop the top $k$ to form a truncated set.
        \State Form target quantiles $y_{j,l}$ using the truncated set.
        \State Update each critic by minimizing the sum of quantile Huber losses.
        \State Update actor using the mean of the critic ensemble estimates.
        \State Update target networks using polyak averaging.
    \EndIf
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{itemize}
    \item \textbf{Distributional Learning:} TQC employs an ensemble of $N$ critic networks, where each network is trained to estimate a specific quantile $\tau_i$ of the return distribution. The critics are trained by minimizing the quantile Huber loss, $L_{QH}$.
    
    \item \textbf{Truncation:} This is the central innovation of TQC. To combat overestimation, the algorithm discards the $k$ largest Q-value estimates from the set of $N$ target values. This truncation removes the most optimistic estimates, which are the primary drivers of overestimation, leading to more conservative and stable target values.
    
    \item \textbf{Critic Update:} The target value for updating the critics is formed using the Bellman equation with the truncated set of next-state Q-values. The overall critic loss is the sum of the quantile losses across all critics:
    \[
    L(\phi) = \sum_{i=1}^{N} \mathbb{E}_{(s,a,r,s') \sim D} \left[ L_{QH}\left(r + \gamma Q_{\text{trunc}}(s', \tilde{a}') - Q_{\phi_i}(s,a) \right) \right]
    \]
    where $Q_{\text{trunc}}$ represents the value derived from the truncated set of target quantiles. The actor is then updated using the mean of all (non-truncated) critic estimates.
\end{itemize}
